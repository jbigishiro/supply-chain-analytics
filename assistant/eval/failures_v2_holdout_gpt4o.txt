Q: What was the month-over-month growth in the number of orders for each month of 2018?
with monthly_orders as (
  select purchase_month, count(distinct order_id) as order_count
  from `supply-chain-analytics-510105.dbt_dev_marts.fact_orders`
  where extract(year from purchase_month) = 2018
  group by purchase_month
),
growth as (
  select 
    purchase_month,
    order_count,
    lag(order_count) over (order by purchase_month) as previous_order_count
  from monthly_orders
)
select 
  purchase_month,
  (order_count - previous_order_count) / previous_order_count as month_over_month_growth
from growth
where previous_order_count is not null
order by purchase_month
